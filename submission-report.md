## Known Bugs or Issues

## Experience

This project was incredibly straightforward, as it asked us to implement the lookup process of a router. This is something we
covered in detail both in class and in the textbook, so it was a simple matter of following the assignment instructions as written.

In terms of design, this program handles the special CIDR boundaries (/0, /31, /32) with explicit conditions. This ensures that
host calculations can be done without undefined behavior in C. /31 uses a point-to-point link, so the first host is the network address
and the last host is the broadcast address. /32 and anything above returns the full mask.

As mentioned above, the forwarding table is parsed using a linear scan to find the longest prefix. Specifically, the function checks
every route to see if they contain the destination and, if so, saves the route with the longest prefix.

#include "lab.h"

#include <stdlib.h>
#include <stdio.h>
#include <string.h>

struct fwd_table {
  route_t *routes;
  size_t count;
  size_t capacity;
};

static bool parse_octet(const char **cursor, uint32_t *value)
{
  const char *p = *cursor;
  uint32_t number = 0;
  size_t digits = 0;

  if (*p == '0') {
    p++;
    digits = 1;
    if (*p >= '0' && *p <= '9') {
      return false;
    }
  } else {
    while (*p >= '0' && *p <= '9') {
      number = number * 10U + (uint32_t)(*p - '0');
      if (number > 255U) {
        return false;
      }
      p++;
      digits++;
    }
  }

  if (digits == 0) {
    return false;
  }

  *cursor = p;
  *value = number;
  return true;
}

bool ip_parse(const char *text, uint32_t *addr)
{
  const char *p = text;
  uint32_t parsed = 0;

  if (text == NULL || addr == NULL) {
    return false;
  }

  for (size_t i = 0; i < 4; i++) {
    uint32_t octet = 0;
    if (!parse_octet(&p, &octet)) {
      return false;
    }
    parsed = (parsed << 8) | octet;
    if (i < 3) {
      if (*p != '.') {
        return false;
      }
      p++;
    }
  }

  if (*p != '\0') {
    return false;
  }

  *addr = parsed;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
  (void)snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
                 (unsigned)((addr >> 24) & 0xffU),
                 (unsigned)((addr >> 16) & 0xffU),
                 (unsigned)((addr >> 8) & 0xffU),
                 (unsigned)(addr & 0xffU));
}

bool cidr_parse(const char *text, cidr_t *block)
{
  const char *slash;
  const char *p;
  char address[IP_STR_LEN];
  uint32_t addr;
  uint32_t prefix = 0;
  size_t address_length;

  if (text == NULL || block == NULL) {
    return false;
  }

  slash = strchr(text, '/');
  if (slash == NULL || strchr(slash + 1, '/') != NULL) {
    return false;
  }

  address_length = (size_t)(slash - text);
  if (address_length == 0 || address_length >= sizeof(address)) {
    return false;
  }
  (void)memcpy(address, text, address_length);
  address[address_length] = '\0';
  if (!ip_parse(address, &addr)) {
    return false;
  }

  p = slash + 1;
  if (*p == '\0' || (*p == '0' && p[1] != '\0')) {
    return false;
  }
  while (*p >= '0' && *p <= '9') {
    prefix = prefix * 10U + (uint32_t)(*p - '0');
    if (prefix > 32U) {
      return false;
    }
    p++;
  }
  if (*p != '\0') {
    return false;
  }

  block->addr = addr;
  block->prefix = (uint8_t)prefix;
  return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN])
{
  char address[IP_STR_LEN];
  ip_format(block.addr, address);
  (void)snprintf(out, CIDR_STR_LEN, "%s/%u", address,
                 (unsigned)(block.prefix > 32U ? 32U : block.prefix));
}

/**
 * @brief Computes the subnet mask for an IPv4 prefix length.
 * @param prefix Prefix length; values above 32 are treated as 32.
 * @return The subnet mask in host byte order.
 */
uint32_t cidr_mask(uint8_t prefix)
{
  if (prefix == 0U) {
    return 0U;
  }
  if (prefix >= 32U) {
    return UINT32_MAX;
  }
  return UINT32_MAX << (32U - prefix);
}

/**
 * @brief Computes the lowest address in a CIDR block.
 * @param block The CIDR block.
 * @return The network address in host byte order.
 */
uint32_t cidr_network(cidr_t block)
{
  return block.addr & cidr_mask(block.prefix);
}

/**
 * @brief Computes the highest address in a CIDR block.
 * @param block The CIDR block.
 * @return The broadcast address in host byte order.
 */
uint32_t cidr_broadcast(cidr_t block)
{
  return cidr_network(block) | ~cidr_mask(block.prefix);
}

/**
 * @brief Computes the first usable host address in a CIDR block.
 * @param block The CIDR block.
 * @return The first usable address, including RFC 3021 handling for /31 and /32.
 */
uint32_t cidr_first_host(cidr_t block)
{
  uint8_t prefix = block.prefix > 32U ? 32U : block.prefix;
  if (prefix >= 31U) {
    return cidr_network(block);
  }
  return cidr_network(block) + 1U;
}

/**
 * @brief Computes the last usable host address in a CIDR block.
 * @param block The CIDR block.
 * @return The last usable address, including RFC 3021 handling for /31 and /32.
 */
uint32_t cidr_last_host(cidr_t block)
{
  uint8_t prefix = block.prefix > 32U ? 32U : block.prefix;
  uint32_t broadcast = cidr_broadcast(block);
  if (prefix >= 31U) {
    return broadcast;
  }
  return broadcast - 1U;
}

/**
 * @brief Counts the addresses available for hosts in a CIDR block.
 * @param block The CIDR block.
 * @return The number of usable host addresses.
 */
uint64_t cidr_host_count(cidr_t block)
{
  uint8_t prefix = block.prefix > 32U ? 32U : block.prefix;
  if (prefix == 31U) {
    return 2U;
  }
  if (prefix == 32U) {
    return 1U;
  }
  return (UINT64_C(1) << (32U - prefix)) - 2U;
}

/**
 * @brief Checks whether an address belongs to a CIDR block.
 * @param block The CIDR block.
 * @param addr The address to test, in host byte order.
 * @return true if addr is contained in block; otherwise false.
 */
bool cidr_contains(cidr_t block, uint32_t addr)
{
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == (block.addr & mask);
}

/**
 * @brief Checks whether a CIDR address has all host bits cleared.
 * @param block The CIDR block to test.
 * @return true if block names a network address; otherwise false.
 */
bool cidr_is_network(cidr_t block)
{
  return block.addr == cidr_network(block);
}

/**
 * @brief Computes one of n equally sized subnets within a CIDR block.
 * @param block The network block to split.
 * @param n Number of subnets; must be a nonzero power of two.
 * @param i Zero-based index of the requested subnet.
 * @param out Destination for the subnet; unchanged when the operation fails.
 * @return true on success; false for invalid input or an impossible split.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
  uint32_t bits = 0;
  uint32_t pieces = n;
  uint32_t new_prefix;
  uint32_t subnet_size;
  cidr_t result;

  if (out == NULL || block.prefix > 32U || !cidr_is_network(block) ||
      n == 0U || (n & (n - 1U)) != 0U || i >= n) {
    return false;
  }

  while (pieces > 1U) {
    pieces >>= 1;
    bits++;
  }
  new_prefix = (uint32_t)block.prefix + bits;
  if (new_prefix > 32U) {
    return false;
  }

  subnet_size = new_prefix == 0U ? 0U : UINT32_C(1) << (32U - new_prefix);
  result.addr = block.addr + i * subnet_size;
  result.prefix = (uint8_t)new_prefix;
  *out = result;
  return true;
}

/**
 * @brief Parses one forwarding-table route line.
 * @param line A route line without comment processing.
 * @param route Destination for the parsed route; unchanged on failure.
 * @return true if line contains one valid route; otherwise false.
 */
bool route_parse(const char *line, route_t *route)
{
  char destination[CIDR_STR_LEN];
  char gateway[IP_STR_LEN];
  char device[DEV_NAME_LEN];
  char extra[2];
  route_t parsed;
  cidr_t block;

  if (line == NULL || route == NULL ||
      sscanf(line, " %18s %15s %15s %1s", destination, gateway, device, extra) != 3) {
    return false;
  }

  if (strcmp(destination, "default") == 0) {
    block.addr = 0U;
    block.prefix = 0U;
  } else if (!cidr_parse(destination, &block) || !cidr_is_network(block)) {
    return false;
  }

  parsed.dest = block;
  if (strcmp(gateway, "direct") == 0) {
    parsed.gateway = 0U;
  } else if (!ip_parse(gateway, &parsed.gateway)) {
    return false;
  }
  (void)memcpy(parsed.dev, device, sizeof(parsed.dev));
  parsed.dev[DEV_NAME_LEN - 1U] = '\0';
  *route = parsed;
  return true;
}

 * @brief Destroys a forwarding table and releases its route storage.

void table_destroy(fwd_table_t *table)
{
  if (table != NULL) {
    free(table->routes);
    free(table);
  }
}

/**
 * @brief Gets the number of routes in a forwarding table.
 * @param table The table to inspect.
 * @return The route count, or zero for NULL.
 */
size_t table_count(const fwd_table_t *table)
{
  return table == NULL ? 0U : table->count;
}

/**
 * @brief Adds a route if its destination is valid and unique.
 * @param table The table to modify.
 * @param route The route to copy into the table.
 * @return true on success; false for invalid, duplicate, or unallocatable input.
 */
bool table_add(fwd_table_t *table, const route_t *route)
{
  route_t *expanded;
  size_t new_capacity;

  if (table == NULL || route == NULL || route->dest.prefix > 32U ||
      !cidr_is_network(route->dest)) {
    return false;
  }
  for (size_t i = 0; i < table->count; i++) {
    if (table->routes[i].dest.addr == route->dest.addr &&
        table->routes[i].dest.prefix == route->dest.prefix) {
      return false;
    }
  }
  if (table->count == table->capacity) {
    new_capacity = table->capacity == 0U ? 4U : table->capacity * 2U;
    expanded = realloc(table->routes, new_capacity * sizeof(*expanded));
    if (expanded == NULL) {
      return false;
    }
    table->routes = expanded;
    table->capacity = new_capacity;
  }
  table->routes[table->count++] = *route;
  return true;
}

/**
 * @brief Loads routes from a forwarding-table file.
 * @param table The table to modify.
 * @param in An open route file.
 * @param bad_line Receives the first invalid line number, or zero for invalid
 *                 arguments or a file-read error.
 * @return true if every non-comment line was loaded; otherwise false.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
  char *line = NULL;
  size_t line_capacity = 0U;
  size_t line_number = 0U;
  bool success = true;

  if (bad_line != NULL) {
    *bad_line = 0U;
  }
  if (table == NULL || in == NULL) {
    return false;
  }

  while (getline(&line, &line_capacity, in) >= 0) {
    route_t route;
    char *comment = strchr(line, '#');
    line_number++;
    if (comment != NULL) {
      *comment = '\0';
    }
    if (line[strspn(line, " \t\r\n")] == '\0') {
      continue;
    }
    if (!route_parse(line, &route) || !table_add(table, &route)) {
      if (bad_line != NULL) {
        *bad_line = line_number;
      }
      success = false;
      break;
    }
  }
  if (success && ferror(in) && bad_line != NULL) {
    *bad_line = 0U;
  }
  free(line);
  return success && !ferror(in);
}

/**
 * @brief Finds the most specific route containing an address.
 * @param table The table to search.
 * @param dest Destination address in host byte order.
 * @return A table-owned longest-prefix match, or NULL if none exists.
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest)
{
  const route_t *best = NULL;

  if (table == NULL) {
    return NULL;
  }
  for (size_t i = 0; i < table->count; i++) {
    const route_t *candidate = &table->routes[i];
    if (cidr_contains(candidate->dest, dest) &&
        (best == NULL || candidate->dest.prefix > best->dest.prefix)) {
      best = candidate;
    }
  }
  return best;
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H

#include "lab.h"

#include <errno.h>
#include <limits.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#ifdef TEST
#define main main_exclude

static void print_usage(void)
{
  printf("Usage: myapp info <address>/<prefix>\n"
         "       myapp split <network>/<prefix> <count>\n"
         "       myapp lookup <table-file> <address>...\n");
}

static bool parse_count(const char *text, uint32_t *count)
{
  char *end;
  unsigned long value;

  if (text == NULL || *text == '\0') {
    return false;
  }
  for (const char *p = text; *p != '\0'; p++) {
    if (*p < '0' || *p > '9') {
      return false;
    }
  }
  errno = 0;
  value = strtoul(text, &end, 10);
  if (errno == ERANGE || *end != '\0' || value > UINT32_MAX) {
    return false;
  }
  *count = (uint32_t)value;
  return true;
}

static int run_info(const char *text)
{
  cidr_t block;
  char address[IP_STR_LEN];
  char mask[IP_STR_LEN];
  char network[IP_STR_LEN];
  char broadcast[IP_STR_LEN];
  char first_host[IP_STR_LEN];
  char last_host[IP_STR_LEN];

  if (text == NULL || !cidr_parse(text, &block)) {
    fprintf(stderr, "myapp: %s: not a valid CIDR block\n",
            text == NULL ? "(null)" : text);
    return 1;
  }
  ip_format(block.addr, address);
  ip_format(cidr_mask(block.prefix), mask);
  ip_format(cidr_network(block), network);
  ip_format(cidr_broadcast(block), broadcast);
  ip_format(cidr_first_host(block), first_host);
  ip_format(cidr_last_host(block), last_host);
  printf("address    %s\n"
         "prefix     /%u\n"
         "netmask    %s\n"
         "network    %s\n"
         "broadcast  %s\n"
         "first host %s\n"
         "last host  %s\n"
         "hosts      %llu\n",
         address, (unsigned)block.prefix, mask, network, broadcast,
         first_host, last_host, (unsigned long long)cidr_host_count(block));
  return 0;
}

static int run_split(const char *text, const char *count_text)
{
  cidr_t block;
  uint32_t count;
  char output[CIDR_STR_LEN];

  if (text == NULL || !cidr_parse(text, &block) ||
      !cidr_is_network(block) || !parse_count(count_text, &count) ||
      count == 0U || (count & (count - 1U)) != 0U) {
    fprintf(stderr, "myapp: invalid network or subnet count\n");
    return 1;
  }
  for (uint32_t i = 0U; i < count; i++) {
    cidr_t subnet;
    if (!cidr_subnet(block, count, i, &subnet)) {
      fprintf(stderr, "myapp: cannot split %s into %u subnets\n",
              text, (unsigned)count);
      return 1;
    }
    cidr_format(subnet, output);
    puts(output);
  }
  return 0;
}

static int run_lookup(const char *filename, int address_count,
                      char *const addresses[])
{
  FILE *in;
  fwd_table_t *table;
  size_t bad_line;
  bool malformed = false;
  bool unreachable = false;

  in = fopen(filename, "r");
  if (in == NULL) {
    fprintf(stderr, "myapp: %s: cannot open table file\n", filename);
    return 2;
  }
  table = table_create();
  if (table == NULL) {
    (void)fclose(in);
    fprintf(stderr, "myapp: could not allocate forwarding table\n");
    return 2;
  }
  if (!table_load(table, in, &bad_line)) {
    if (bad_line != 0U) {
      fprintf(stderr, "myapp: %s:%zu: not a valid route\n",
              filename, bad_line);
    } else {
      fprintf(stderr, "myapp: %s: could not read table file\n", filename);
    }
    (void)fclose(in);
    table_destroy(table);
    return 2;
  }
  (void)fclose(in);

  for (int i = 0; i < address_count; i++) {
    uint32_t address;
    const route_t *route;
    char formatted[IP_STR_LEN];
    if (!ip_parse(addresses[i], &address)) {
      fprintf(stderr, "myapp: %s: not a valid address\n", addresses[i]);
      malformed = true;
      continue;
    }
    ip_format(address, formatted);
    route = table_lookup(table, address);
    if (route == NULL) {
      printf("%s unreachable\n", formatted);
      unreachable = true;
    } else if (route->gateway == 0U) {
      printf("%s dev %s\n", formatted, route->dev);
    } else {
      char gateway[IP_STR_LEN];
      ip_format(route->gateway, gateway);
      printf("%s via %s dev %s\n", formatted, gateway, route->dev);
    }
  }
  table_destroy(table);
  if (malformed) {
    return 1;
  }
  return unreachable ? 3 : 0;
}

int main(int argc, char *argv[])
{
  if (argc == 1) {
    print_usage();
    return 0;
  }
  if (strcmp(argv[1], "info") == 0 && argc == 3) {
    return run_info(argv[2]);
  }
  if (strcmp(argv[1], "split") == 0 && argc == 4) {
    return run_split(argv[2], argv[3]);
  }
  if (strcmp(argv[1], "lookup") == 0 && argc >= 4) {
    return run_lookup(argv[2], argc - 3, &argv[3]);
  }
  fprintf(stderr, "myapp: invalid command line\n");
  print_usage();
  return 1;
}
